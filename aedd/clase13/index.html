CalcularLiquidacion

8 pts

Defina e implemente la función CalcularLiquidacion(...) que reciba la matriz, el vector de nombres, el vector de tarifas y un string con el nombre de un chofer (ingresado por teclado).

La función debe buscar a ese chofer y retornar el monto total a cobrar en el mes (calculado como: total de paquetes × tarifa base). Si el chofer no existe, debe retornar -1.

Puesta en común

d

Solución: CalcularLiquidacion


float CalcularLiquidacion(int envios[][DIAS], string nombres[],
        float tarifas[], string choferBuscado) {
    int indiceChofer = -1;
    int i = 0;
    bool encontrado = false;
    while (i < CHOFERES && !encontrado) {
        if (nombres[i] == choferBuscado) {
            indiceChofer = i;
            encontrado = true;
        }
        i++;
    }

    int totalPaquetes = 0;
    if (indiceChofer == -1) {
        totalPaquetes = -1;
    } else {
        for (int c = 0; c < DIAS; c++) {
            totalPaquetes += envios[indiceChofer][c];
        }
        totalPaquetes = totalPaquetes * tarifas[indiceChofer];
    }
    return totalPaquetes;
}

int ChoferesInactivosDia(int envios[][DIAS], int D, int f) {

#include <iostream>
#include <string>
using namespace std;

const int CHOFERES = 25;
const int DIAS = 31;

string MejorChofer(int envios[][DIAS], string nombres[]) {
    int maxPaquetes = -1;
    string mejorNombre = "";
    for (int f = 0; f < CHOFERES; f++) {
        int sumaActual = 0;
        for (int c = 0; c < DIAS; c++) {
            sumaActual += envios[f][c];
        }
        if (sumaActual > maxPaquetes) {
            maxPaquetes = sumaActual;
            mejorNombre = nombres[f];
        }
    }
    return mejorNombre;
}

int DiasBajaProduccion(int envios[][DIAS]) {
    int diasBajos = 0;
    for (int c = 0; c < DIAS; c++) {
        int sumaDia = 0;
        for (int f = 0; f < CHOFERES; f++) {
            sumaDia += envios[f][c];
        }
        if (sumaDia < 100) {
            diasBajos++;
        }
    }
    return diasBajos;
}

float CalcularLiquidacion(int envios[][DIAS], string nombres[],
        float tarifas[], string choferBuscado) {
    int indiceChofer = -1;
    int i = 0;
    bool encontrado = false;
    while (i < CHOFERES && !encontrado) {
        if (nombres[i] == choferBuscado) {
            indiceChofer = i;
            encontrado = true;
        }
        i++;
    }
    int totalPaquetes = 0;
    if (indiceChofer == -1) {
        totalPaquetes = -1;
    } else {
        for (int c = 0; c < DIAS; c++) {
            totalPaquetes += envios[indiceChofer][c];
        }
        totalPaquetes = totalPaquetes * tarifas[indiceChofer];
    }
    return totalPaquetes;
}

int ChoferesInactivosDia(int envios[][DIAS], int D, int f) {
    if (f == CHOFERES) {
        return 0;
    }
    int inactivo = 0;
    if (envios[f][D] == 0) {
        inactivo = 1;
    }
    return inactivo + ChoferesInactivosDia(envios, D, f + 1);
}

int main() {
    int matrizEnvios[CHOFERES][DIAS];
    string nombresChoferes[CHOFERES];
    float tarifasChoferes[CHOFERES];

    cout << MejorChofer(matrizEnvios, nombresChoferes) << endl;
    cout << DiasBajaProduccion(matrizEnvios) << endl;

    string nombreBuscado;
    cin >> nombreBuscado;
    float pago = CalcularLiquidacion(matrizEnvios, nombresChoferes,
        tarifasChoferes, nombreBuscado);
    cout << pago << endl;

    cout << ChoferesInactivosDia(matrizEnvios, 15, 0) << endl;
    return 0;
}
