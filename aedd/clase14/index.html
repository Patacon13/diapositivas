Premios()

8 pts

Asumiendo que los datos de todas las estructuras se encuentran debidamente cargados, codifique la función Premios(), que reciba la MatrizTecno, un número Level de nivel de competencia, y un entero position en 1..3, y retorne la siguiente información del equipo que obtuvo la posición position en la competencia de nivel Level:

  • Nro. de Equipo
  • Nombre de Fantasía
  • Porcentaje aceptados/enviados

Puesta en común

c

Solución: Premios()


struct InfoPremio {
    int nroEquipo;
    string nombreFantasia;
    float porcentajeAceptados;
};

InfoPremio Premios(InfoTecno matrizTecno[][MAX_COL_TECNO], Equipo equipos[], int level, int position) {
    InfoPremio resultado;
    int f = level - 1;
    int c = position - 1;

    int nroEq = matrizTecno[f][c].nroEquipo;
    resultado.nroEquipo = nroEq;
    resultado.nombreFantasia = equipos[nroEq].nombreFantasia;

    if (matrizTecno[f][c].envios > 0) {
        resultado.porcentajeAceptados = ((float)matrizTecno[f][c].aceptados / matrizTecno[f][c].envios) * 100.0f;
    } else {
        resultado.porcentajeAceptados = 0.0f;
    }

    return resultado;
}

UTNSantaFeesTop()

8 pts

Codifique la función UTNSantaFeesTop() que determine si se cumple la siguiente condición: en los 3 niveles de competencia más altos (el nivel de fila 0 corresponde a secundaria) la UTN Santa Fe obtuvo alguno de los tres primeros puestos.

Los equipos se ordenan según resuelvan mayor cantidad de problemas, y en caso de empate resulta ganador el equipo que hizo el último envío en menor cantidad de tiempo.
La MatrizTecno se ha ordenado de tal forma que todas las filas contienen a los tres primeros equipos de ese nivel en sus tres primeras posiciones (ordenados del 1° al 3°).

Puesta en común

d

Solución: UTNSantaFeesTop()


bool UTNSantaFeesTop(InfoTecno matrizTecno[][MAX_COL_TECNO], Equipo equipos[]) {
    bool topEnTodos = true;
    int f = 1;

    while (f < NIVELES && topEnTodos) {
        bool podioNivel = false;
        int c = 0;

        while (c < 3 && !podioNivel) {
            int nroEq = matrizTecno[f][c].nroEquipo;
            if (equipos[nroEq].institucion == "UTN Santa Fe") {
                podioNivel = true;
            }
            c++;
        }

        if (!podioNivel) {
            topEnTodos = false;
        }
        f++;
    }

    return topEnTodos;
}

RankingGlobal()

8 pts

Codifique la función RankingGlobal() que obtenga los 3 primeros puestos entre todos los equipos en las tres primeras posiciones de cada nivel, según el valor de Enviados/Aceptados. La función debe retornar los 3 números que identifican a estos equipos.

Puesta en común

e

Solución: RankingGlobal()


struct Candidato {
    int nroEquipo;
    float ratio;
};

struct PodioGlobal {
    int primero;
    int segundo;
    int tercero;
};

PodioGlobal RankingGlobal(InfoTecno matrizTecno[][MAX_COL_TECNO]) {
    Candidato finalistas[12];
    int k = 0;

    for (int f = 0; f < NIVELES; f++) {
        for (int c = 0; c < 3; c++) {
            finalistas[k].nroEquipo = matrizTecno[f][c].nroEquipo;
            if (matrizTecno[f][c].aceptados > 0) {
                finalistas[k].ratio = (float)matrizTecno[f][c].envios / matrizTecno[f][c].aceptados;
            } else {
                finalistas[k].ratio = 0.0f;
            }
            k++;
        }
    }

    for (int i = 0; i < 3; i++) {
        int posMax = i;
        for (int j = i + 1; j < 12; j++) {
            if (finalistas[j].ratio > finalistas[posMax].ratio) {
                posMax = j;
            }
        }
        Candidato aux = finalistas[i];
        finalistas[i] = finalistas[posMax];
        finalistas[posMax] = aux;
    }

    PodioGlobal podio;
    podio.primero = finalistas[0].nroEquipo;
    podio.segundo = finalistas[1].nroEquipo;
    podio.tercero = finalistas[2].nroEquipo;
    return podio;
}
