¿Qué es un TDA?

Definición

Un Tipo de Dato Abstracto (TDA) es un modelo matemático compuesto por un conjunto de valores y un conjunto de operaciones primitivas definidas sobre esos valores.

Entradas

CAJA NEGRA

Ocultamiento de Información
(Representación interna privada)

Salidas

Interfaz Pública

Define qué operaciones se pueden realizar (visibles para el usuario/cliente).

Implementación Privada

Define cómo se almacenan los datos y cómo funcionan los algoritmos internos.

Estructura Modular en C++

archivo.h

Interfaz / Declaración

  • Definición del struct o tipo de dato.
  • Prototipos de funciones primitivas.
  • Documentación de uso (pre/post-condiciones).

archivo.cpp

Implementación

  • Código fuente de cada operación primitiva.
  • Incluye #include "archivo.h".
  • Oculta los detalles al programa cliente.

main.cpp

Programa Cliente

  • Utiliza el TDA a través de su interfaz pública.
  • No accede a campos internos si se desea encapsular.
  • Crea y manipula instancias del tipo.

TDA Círculo & TDA Cuadrado

circulo.h / circulo.cpp 

// === circulo.h ===
const float PI = 3.14159265f;
struct Circulo { float radio; };
Circulo crear_circulo(float r);
float perimetro(Circulo c);
float area(Circulo c);

// === circulo.cpp ===
#include "circulo.h"
Circulo crear_circulo(float r) {
    Circulo c; c.radio = r; return c;
}
float perimetro(Circulo c) {
    return 2.0f * PI * c.radio;
}
float area(Circulo c) {
    return PI * c.radio * c.radio;
}
cuadrado.h / cuadrado.cpp 

// === cuadrado.h ===
struct Cuadrado { float lado; };
Cuadrado crear_cuadrado(float l);
float perimetro(Cuadrado c);
float area(Cuadrado c);

// === cuadrado.cpp ===
#include "cuadrado.h"
Cuadrado crear_cuadrado(float l) {
    Cuadrado c; c.lado = l; return c;
}
float perimetro(Cuadrado c) {
    return 4.0f * c.lado;
}
float area(Cuadrado c) {
    return c.lado * c.lado;
}

TDA Rectángulo & TDA Triángulo

rectangulo.h / rectangulo.cpp 

// === rectangulo.h ===
struct Rectangulo { float base, altura; };
Rectangulo crear_rectangulo(float b, float h);
float perimetro(Rectangulo r);
float area(Rectangulo r);

// === rectangulo.cpp ===
#include "rectangulo.h"
Rectangulo crear_rectangulo(float b, float h) {
    Rectangulo r; r.base = b; r.altura = h; return r;
}
float perimetro(Rectangulo r) {
    return 2.0f * (r.base + r.altura);
}
float area(Rectangulo r) {
    return r.base * r.altura;
}
triangulo.h / triangulo.cpp 

// === triangulo.h ===
struct Triangulo { float base, altura, l1, l2, l3; };
Triangulo crear_triangulo(float b, float h,
                          float l1, float l2, float l3);
float perimetro(Triangulo t);
float area(Triangulo t);

// === triangulo.cpp ===
#include "triangulo.h"
Triangulo crear_triangulo(float b, float h,
                          float l1, float l2, float l3) {
    Triangulo t; t.base = b; t.altura = h;
    t.l1 = l1; t.l2 = l2; t.l3 = l3; return t;
}
float perimetro(Triangulo t) {
    return t.l1 + t.l2 + t.l3;
}
float area(Triangulo t) {
    return (t.base * t.altura) / 2.0f;
}

    return 0;
}

Solución: TDA Hora - Implementación (.cpp)


Hora ponerAHora(int h, int m, int s) {
    Hora nueva;
    if (h >= 0 && h < 24) nueva.hora = h;
    else nueva.hora = 0;

    if (m >= 0 && m < 60) nueva.min = m;
    else nueva.min = 0;

    if (s >= 0 && s < 60) nueva.seg = s;
    else nueva.seg = 0;

    return nueva;
}

void imprimirDosDigitos(int n) {
    if (n < 10) cout << "0";
    cout << n;
}

void mostrar24(Hora h) {
    imprimirDosDigitos(h.hora); cout << ":";
    imprimirDosDigitos(h.min);  cout << ":";
    imprimirDosDigitos(h.seg);  cout << endl;
}

void mostrar12(Hora h) {
    int h12 = h.hora;
    string sufijo = " a.m.";
    if (h.hora >= 12) {
        sufijo = " p.m.";
        if (h.hora > 12) h12 = h.hora - 12;
    } else {
        if (h.hora == 0) h12 = 12;
    }
    imprimirDosDigitos(h12);   cout << ":";
    imprimirDosDigitos(h.min); cout << ":";
    imprimirDosDigitos(h.seg); cout << sufijo << endl;
}

void adelantar(Hora &h, int minutos) {
    int total = (h.hora * 60 + h.min + minutos) % (24 * 60);
    if (total < 0) total += (24 * 60);
    h.hora = total / 60;
    h.min = total % 60;
}

void atrasar(Hora &h, int minutos) {
    adelantar(h, -minutos);
}

int diferenciaEnMinutos(Hora h1, Hora h2) {
    int diff = (h1.hora * 60 + h1.min) - (h2.hora * 60 + h2.min);
    if (diff < 0) diff = -diff;
    return diff;
}

Solución: TDA Hora - Programa Cliente (main.cpp)


#include <iostream>
using namespace std;

int main() {
    Hora h = ponerAHora(14, 30, 0);
    cout << "Hora en 24hs: "; mostrar24(h);
    cout << "Hora en 12hs: "; mostrar12(h);

    cout << endl << "Adelantamos 45 minutos:" << endl;

    cout << "Hora en 12hs: "; mostrar12(h);

    cout << endl << "Atrasamos 120 minutos:" << endl;
    atrasar(h, 120);
    cout << "Hora en 24hs: "; mostrar24(h);

    Hora h2 = ponerAHora(17, 15, 0);
    cout << endl << "Diferencia con 17:15: " << diferenciaEnMinutos(h, h2) << " minutos." << endl;

    return 0;
}

TDA Pila: Acceso LIFO

Estructura Lineal

Estructura LIFO (Last-In, First-Out)

Solo se tiene acceso al último elemento ingresado: TOS (Top of Stack / Cima).

Push (Apilar) Pop (Desapilar)
Dato Base [0]
Dato Intermedio [1]
Dato Cima [2]
TOS (Cima)

Analogía: Pila de platos (se coloca y retira siempre por arriba).

En la Memoria: Call Stack

El procesador organiza las llamadas a funciones y sus variables locales utilizando exactamente una pila:

0xFFFF Función activa (Cima / TOS)
0xFFE0 Variables locales de funcion()
0xFA00 Frame de llamada de main()
0x0000 Variables globales y código

Al invocar una función, se apila su marco; al retornar con return, se desapila.

TDA Pila: Especificación

Operaciones Primitivas

Implementar el TDA Pila de Enteros sobre un vector estático de tamaño TF y su índice de tope, según la teoría de cátedra:

  • void inicializar(Pila &P): inicializa la pila con tope = -1.
  • bool agregar(Pila &P, int valor): inserta en el tope. Retorna false si la pila está llena (overflow).
  • bool sacar(Pila &P, int &valor): retira del tope asignando el dato a valor. Retorna false si está vacía (underflow).
  • bool vacia(const Pila P): retorna true si tope == -1.
  • void mostrarPila(const Pila P): imprime el contenido y el tope actual.
  • void invertirPila(Pila &P): invierte el orden usando una pila auxiliar (LIFO → LIFO).

Solución: TDA Pila - Interfaz (.h)


const int TF = 100;

// Estructura de la Pila sobre Vector
struct Pila {
    int info[TF];  // vector para almacenar los valores
    int tope;      // subíndice del último elemento insertado
};

// Prototipos de operaciones primitivas
void inicializar(Pila &P);
bool agregar(Pila &P, int valor);
bool sacar(Pila &P, int &valor);
bool vacia(const Pila P);
void mostrarPila(const Pila P);
void invertirPila(Pila &P);
3

Solución: TDA Pila - Implementación (.cpp)


#include "pila.h"

void inicializar(Pila &P) { P.tope = -1; }

bool vacia(const Pila P) { return (P.tope == -1); }

bool agregar(Pila &P, int valor) {
    if (P.tope == TF - 1) return false; // Overflow
    P.tope++;
    P.info[P.tope] = valor;
    return true;
}

bool sacar(Pila &P, int &valor) {
    if (vacia(P)) return false; // Underflow
    valor = P.info[P.tope];
    P.tope--;
    return true;
}

void mostrarPila(const Pila P) {
    cout << "Pila = [ ";
    for (int i = 0; i <= P.tope; i++)
        cout << P.info[i] << " ";
    cout << "] , Tope = " << P.tope << endl;
}

void invertirPila(Pila &P) {
    Pila Aux; int val;
    inicializar(Aux);
    while (!vacia(P)) {
        sacar(P, val); agregar(Aux, val);
    }
    P = Aux;
}

Solución: TDA Pila - Cliente (usaTDAPila.cpp)


#include <iostream>
#include "pila.h"
using namespace std;

int main() {
    Pila P; int aux;
    inicializar(P);
    mostrarPila(P);

    agregar(P, 3);
    agregar(P, 8);
    agregar(P, 5);
    mostrarPila(P);

    cout << endl << "Invertir Pila:" << endl;
    invertirPila(P);
    mostrarPila(P);

    cout << endl << "Desapilando elementos en orden LIFO:" << endl;
    while (sacar(P, aux)) {
        cout << "Se retiro: " << aux << " | ";
        mostrarPila(P);
    }
    if (!sacar(P, aux)) cout << "Underflow = Pila vacia" << endl;

    return 0;
}

TDA Cola: Acceso FIFO

Estructura Lineal

Estructura FIFO (First-In, First-Out)

Inserción por un extremo (Final) y extracción por el otro extremo (Frente).

Desencolar (Pop / Frente) Encolar (Push / Final)
D1 Frente
D2 En cola
D3 En cola
D4 Final

El primer elemento en ingresar es el primero en ser atendido y salir.

En Sistemas: Peticiones al Servidor

Modelo clásico de atención de solicitudes en redes, colas de impresión y eventos de usuario:

C3 C2 C1
SERVIDOR
¡Atendiendo C1!

Usos frecuentes: Cola de impresión (Spooler), buffer del teclado en el SO, tareas en background.

¿Por qué Colas Circulares?

Cola Lineal sobre Vector

  • Opción 1 (Corrimiento): Cada vez que se desencola, desplazar los $N$ elementos restantes hacia la izquierda. Costo: $O(N)$ por extracción.
  • Opción 2 (Solo avanzar frente): Si solo se incrementa Primero, las posiciones iniciales quedan inutilizadas. La cola se declara llena aun teniendo espacio libre.

Cola Circular ($O(1)$)

  • Aritmética modular: Los índices avanzan con (pos + 1) % TF. Al llegar a TF-1, pegan la vuelta al índice 0.
  • Costo $O(1)$ óptimo: Encolar y desencolar sin realizar ningún corrimiento en memoria.
  • Condición de Cola Llena: (Ultimo + 1) % TF == Primero (se reserva 1 casillero para diferenciar "llena" de "vacía").

Convención de Cátedra: La cola vacía ocurre cuando Primero == Ultimo. Por ello, en un vector de tamaño TF, la capacidad máxima de datos útiles es TF - 1.

4

Solución: Cola Circular - Interfaz (.h)


const int TF = 5; // Capacidad útil: TF - 1 = 4

// Definición de la estructura para la cola circular
struct Cola {
    int info[TF];         // vector para almacenar los valores
    int Primero, Ultimo;  // subíndices de frente y final
};

// Prototipos de operaciones primitivas
void inicializar(Cola &C);
bool encolar(Cola &C, int valor);
bool sacar(Cola &C, int &valor);
bool vacia(const Cola C);
void mostrarCola(const Cola C);
4

Solución: Cola Circular - Implementación (.cpp)


#include "cola.h"

void inicializar(Cola &C) { C.Primero = C.Ultimo = 0; }

bool vacia(const Cola C) { return (C.Primero == C.Ultimo); }

bool encolar(Cola &C, int valor) {
    if ((C.Ultimo + 1) % TF == C.Primero)
        return false; // Overflow: cola llena
    C.info[C.Ultimo] = valor;
    C.Ultimo = (C.Ultimo + 1) % TF;
    return true;
}

bool sacar(Cola &C, int &valor) {
    if (vacia(C)) return false; // Underflow: cola vacía
    valor = C.info[C.Primero];
    C.Primero = (C.Primero + 1) % TF;
    return true;
}

void mostrarCola(const Cola C) {
    cout << "Cola = [ ";
    for (int i = C.Primero; i != C.Ultimo; i = (i + 1) % TF)
        cout << C.info[i] << " ";
    cout << "] , Primero = " << C.Primero << " , Ultimo = " << C.Ultimo << endl;
}

Solución: Cola Circular - Cliente (usaTDACola.cpp)


#include <iostream>
#include "cola.h"
using namespace std;

int main() {
    Cola C; int aux;
    inicializar(C);
    mostrarCola(C);

    encolar(C, 3); encolar(C, 8); encolar(C, 5); encolar(C, 1);
    mostrarCola(C);

    if (!encolar(C, 6)) cout << ">> Overflow: Cola llena (limite 4 elementos)" << endl;

    if (sacar(C, aux)) cout << ">> Se retiro elemento frontal = " << aux << endl;
    mostrarCola(C);

    if (encolar(C, 99)) cout << ">> Se encolo 99 en posicion circular libre!" << endl;
    mostrarCola(C);

    return 0;
}
